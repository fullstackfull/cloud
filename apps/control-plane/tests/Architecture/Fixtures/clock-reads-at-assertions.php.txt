final class WalkTest extends TestCase
{
    // Any closure is walked where it is written.
    #[Test]
    public function found_in_another_closure(): void
    {

    // An arrow function's pin does not leak out either.

    {

        $this->assertEquals(now(), $row->at);
    }

    // A call's receiver is walked ...
    #[Test]
    public function clean_a_pin_in_a_receiver(): void
    {
        $this->freezeSecond()->toDateString();
        $this->assertEquals(now(), $row->at);
    }

    // ... and its arguments ...
    #[Test]
    public function clean_a_pin_in_an_argument(): void
    {
        $this->run($this->freezeSecond());
        $this->assertEquals(now(), $row->at);
    }

    // ... before an assertion is judged.
    #[Test]
    public function clean_an_assertion_whose_arguments_pin(): void
    {
        $this->assertEquals($this->freezeSecond(), now());
    }

    // A test-now scope's callback is its second argument.
    #[Test]
    public function found_native_inside_a_test_now_scope(): void
    {
        Carbon::withTestNow('2026-01-01 00:00:00', fn () => $this->assertSame(time(), $this->row->stamp));
    }

    // freezeTime(null) has no callback: it pins.
    #[Test]
    public function clean_freeze_given_a_null_callback(): void
    {
        $this->freezeTime(null);
        $this->assertEquals(now(), $row->at);
    }

    // A nullsafe call on $this pins.
    #[Test]
    public function clean_a_nullsafe_pin(): void
    {
        $this?->freezeSecond();
        $this->assertEquals(now(), $row->at);
    }

    // A nullsafe predicate method is an equality.
    #[Test]
    public function found_a_nullsafe_predicate_method(): void
    {
        $this->assertTrue($row->at?->eq(now()));
    }

    // Implicit methods read the clock only when given nothing.
    #[Test]
    public function clean_a_diff_between_two_stored_dates(): void
    {
        $this->assertSame(7, $row->ends_at->diffInDays($row->starts_at));
    }

    // Freezing reaches `age`.
    #[Test]
    public function clean_age_on_a_frozen_clock(): void
    {
        $this->freezeSecond();
        $this->assertSame(30, $row->born->age);
    }

    // Only the first argument is read as a literal.
    #[Test]
    public function clean_a_relative_word_that_is_not_the_first_argument(): void
    {
        $this->assertEquals(Carbon::parse('2026-01-01', 'today'), $row->at);
        $this->assertEquals(new Carbon('2026-01-01', 'today'), $row->at);
    }

    // A named argument of an assertion that compares nothing is not compared.
    #[Test]
    public function clean_a_named_argument_of_a_non_equality_assertion(): void
    {
        $this->assertNotNull(actual: now());
    }

    // A predicate's second argument is its message.
    #[Test]
    public function clean_a_predicate_message(): void
    {
        $this->assertTrue($ok, $row->day === now()->toDateString());
    }

    // Carbon::make('') is null; a constructor given '' is now.
    #[Test]
    public function clean_make_of_a_blank_literal(): void
    {

    }

    #[Test]
    public function found_a_constructor_of_a_blank_literal(): void

    // PHPUnit's function-style assertions are assertions.
    #[Test]

    // PHP's function names are case-insensitive.
    #[Test]

    }

    // An unpacked argument is not read (a blind spot the docblock names).
    #[Test]
    public function clean_an_unpacked_argument(): void
