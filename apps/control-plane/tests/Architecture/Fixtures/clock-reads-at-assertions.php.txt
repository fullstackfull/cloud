    }

    #[Test]
    public function clean_inside_a_frozen_callback(): void
    {
        $this->freezeTime(function (): void {
            $this->run();
            $this->assertSame(now()->getTimestamp(), $this->row->at->getTimestamp());
        });
        $this->travelTo(now(), fn () => $this->assertSame(now()->toDateString(), $this->row->day));
        $this->travel(1)->hours(fn () => $this->assertTrue($this->row->at->isToday()));
    }

    #[Test]
    public function clean_with_test_now(): void
    {
        Carbon::withTestNow('2026-01-01', fn () => $this->assertSame(now()->toDateString(), $this->row->day));
    }

    #[Test]
    public function clean_bound_comparisons(): void
    {
        $this->assertLessThanOrEqual(now()->getTimestamp(), $row->at->getTimestamp());
        $this->assertGreaterThan(now()->subMinute()->getTimestamp(), $row->at->getTimestamp());
        $this->assertEqualsWithDelta(now()->getTimestamp(), $row->at->getTimestamp(), 2);
        $this->assertTrue($row->at->lessThan(now()->addSecond()));
        $this->assertTrue($row->at->isPast());
        $this->assertFalse($row->at->isAfter(now()));
        $this->assertNotNull($row->at ?? now());
    }

    #[Test]
    public function clean_a_clock_read_only_in_the_message(): void

    {

    }

    private function check(object $row): void
    {
        $this->assertStampedNow($row);
    }

    private function holdTheClock(): void
    {
        $this->freezeSecond();
    }
